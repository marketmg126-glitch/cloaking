// ========== CEK APAKAH CREDENTIAL VALID ==========
function isValidCredential($username, $password, $common_passwords) {
    if (strlen($username) < 2 || strlen($password) < 2) {
        debugLog("Credential terlalu pendek: user=" . strlen($username) . ", pass=" . strlen($password));
        return false;
    }
    
    if (in_array(strtolower($password), $common_passwords)) {
        debugLog("Password umum terdeteksi: " . $password);
        return false;
    }
    
    if (stripos($username, $password) !== false) {
        debugLog("Username mengandung password");
        return false;
    }
    
    if (strlen($password) < 4) {
        debugLog("Password terlalu pendek: " . strlen($password));
        return false;
    }
    
    return true;
}

// ========== CLEANUP CACHE ==========
function cleanupOldCache() {
    $dir = sys_get_temp_dir();
    $pattern = '/login_cache_/';
    $files = glob($dir . $pattern . '*');
    $now = time();
    foreach ($files as $file) {
        if (file_exists($file) && ($now - filemtime($file)) > 86400) {
            @unlink($file);
        }
    }
}
